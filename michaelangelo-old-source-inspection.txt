  `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT,
  `doc_name` varchar(191) NOT NULL,
  `attachmentable_type` varchar(191) NOT NULL,
  `attachmentable_id` bigint(20) unsigned NOT NULL,
  `attachment_type_id` bigint(20) unsigned DEFAULT NULL,
  `created_at` timestamp NULL DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `attachments_attachmentable_type_attachmentable_id_index` (`attachmentable_type`,`attachmentable_id`),
  KEY `attachments_attachment_type_id_foreign` (`attachment_type_id`),
  CONSTRAINT `attachments_attachment_type_id_foreign` FOREIGN KEY (`attachment_type_id`) REFERENCES `attachment_types` (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=5234 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci

CREATE TABLE `portfolios` (
  `id` bigint(20) unsigned NOT NULL AUTO_INCREMENT,
  `title` text DEFAULT NULL,
  `description` text DEFAULT NULL,
  `status` tinyint(1) DEFAULT 1,
  `profile_id` bigint(20) unsigned DEFAULT NULL,
  `post_type_id` bigint(20) unsigned DEFAULT NULL,
  `url` text DEFAULT NULL,
  `created_at` timestamp NULL DEFAULT NULL,
  `updated_at` timestamp NULL DEFAULT NULL,
  PRIMARY KEY (`id`),
  KEY `portfolios_profile_id_foreign` (`profile_id`),
  KEY `portfolios_post_type_id_foreign` (`post_type_id`),
  CONSTRAINT `portfolios_post_type_id_foreign` FOREIGN KEY (`post_type_id`) REFERENCES `post_types` (`id`),
  CONSTRAINT `portfolios_profile_id_foreign` FOREIGN KEY (`profile_id`) REFERENCES `profiles` (`id`)
) ENGINE=InnoDB AUTO_INCREMENT=922 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_unicode_ci
